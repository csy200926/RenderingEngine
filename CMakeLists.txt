cmake_minimum_required(VERSION 3.24)
cmake_policy(SET CMP0091 NEW)
project(MyEngine VERSION 1.0 LANGUAGES C CXX)
if(NOT WIN32 OR NOT MSVC)
    message(FATAL_ERROR "Install Visual Studio 2022 Desktop development with C++ and a Windows SDK. This project uses Win32 and desktop graphics APIs.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(ASSIMP_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(ASSIMP_BUILD_ASSIMP_TOOLS OFF CACHE BOOL "" FORCE)
set(ASSIMP_BUILD_SAMPLES OFF CACHE BOOL "" FORCE)
set(ASSIMP_INSTALL OFF CACHE BOOL "" FORCE)
set(ASSIMP_WARNINGS_AS_ERRORS OFF CACHE BOOL "" FORCE)
set(ASSIMP_BUILD_ZLIB ON CACHE BOOL "" FORCE)
set(ASSIMP_BUILD_ALL_IMPORTERS_BY_DEFAULT OFF CACHE BOOL "" FORCE)
set(ASSIMP_BUILD_OBJ_IMPORTER ON CACHE BOOL "" FORCE)
set(ASSIMP_BUILD_FBX_IMPORTER ON CACHE BOOL "" FORCE)
set(ASSIMP_NO_EXPORT ON CACHE BOOL "" FORCE)
set(ASSIMP_BUILD_ALL_EXPORTERS_BY_DEFAULT OFF CACHE BOOL "" FORCE)
set(ASSIMP_BUILD_DRACO OFF CACHE BOOL "" FORCE)
set(ASSIMP_INJECT_DEBUG_POSTFIX OFF CACHE BOOL "" FORCE)
set(ASSIMP_BUILD_USE_CCACHE OFF CACHE BOOL "" FORCE)
set(USE_STATIC_CRT ON CACHE BOOL "" FORCE)
set(ASSIMP_BUILD_DOCS OFF CACHE BOOL "" FORCE)
set(ASSIMP_HUNTER_ENABLED OFF CACHE BOOL "" FORCE)
if(POLICY CMP0175)
    set(CMAKE_POLICY_DEFAULT_CMP0175 OLD)
endif()
add_subdirectory(ThirdParty/assimp-5.4.3 EXCLUDE_FROM_ALL)
file(GLOB LUA_SOURCES CONFIGURE_DEPENDS ThirdParty/lua-5.4.8/src/*.c)
list(FILTER LUA_SOURCES EXCLUDE REGEX "/(lua|luac)\\.c$")
add_library(MyLua STATIC ${LUA_SOURCES})
target_include_directories(MyLua PUBLIC ThirdParty/lua-5.4.8/src)
target_compile_definitions(MyLua PRIVATE _CRT_SECURE_NO_WARNINGS)
file(GLOB_RECURSE ENGINE_SOURCES CONFIGURE_DEPENDS Engine/*.cpp Rendering/*.cpp Platform/*.cpp)
list(APPEND ENGINE_SOURCES ThirdParty/stb_image.c)
add_library(MyEngine STATIC ${ENGINE_SOURCES})
target_include_directories(MyEngine PUBLIC . ThirdParty ThirdParty/glm PRIVATE ThirdParty/Vulkan/Include)
target_compile_definitions(MyEngine PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX GLM_FORCE_RADIANS _CRT_SECURE_NO_WARNINGS)
target_compile_options(MyEngine PRIVATE /W3 /utf-8 /MP)
target_link_libraries(MyEngine PUBLIC MyLua assimp opengl32 d3d11 d3d12 dxgi d3dcompiler user32 gdi32 shell32 ole32)
file(GLOB GAME_SOURCES CONFIGURE_DEPENDS Game/*.cpp)
add_executable(Renderer ${GAME_SOURCES})
# 固定在工程内的 DXC 把 HLSL 编成 SPIR-V；不用查找 PATH 或 VULKAN_SDK。
set(LOCAL_DXC "${CMAKE_CURRENT_SOURCE_DIR}/ThirdParty/Vulkan/Tools/dxc.exe")
set(VULKAN_SHADER_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/Assets/Shaders/Vulkan/Default.hlsl")
set(VULKAN_SHADER_OUTPUT "${CMAKE_BINARY_DIR}/Shaders/Vulkan")
add_custom_command(OUTPUT "${VULKAN_SHADER_OUTPUT}/Default.vert.spv" "${VULKAN_SHADER_OUTPUT}/Default.frag.spv"
    COMMAND ${CMAKE_COMMAND} -E make_directory "${VULKAN_SHADER_OUTPUT}"
    COMMAND "${LOCAL_DXC}" -spirv -fspv-target-env=vulkan1.1 -fvk-use-dx-layout -fvk-b-shift 0 0 -fvk-t-shift 1 0 -fvk-s-shift 7 0 -fvk-invert-y -T vs_6_0 -E VSMain -Fo "${VULKAN_SHADER_OUTPUT}/Default.vert.spv" "${VULKAN_SHADER_SOURCE}"
    COMMAND "${LOCAL_DXC}" -spirv -fspv-target-env=vulkan1.1 -fvk-use-dx-layout -fvk-b-shift 0 0 -fvk-t-shift 1 0 -fvk-s-shift 7 0 -T ps_6_0 -E PSMain -Fo "${VULKAN_SHADER_OUTPUT}/Default.frag.spv" "${VULKAN_SHADER_SOURCE}"
    DEPENDS "${VULKAN_SHADER_SOURCE}" "${LOCAL_DXC}" VERBATIM)
add_custom_target(VulkanShaders DEPENDS "${VULKAN_SHADER_OUTPUT}/Default.vert.spv" "${VULKAN_SHADER_OUTPUT}/Default.frag.spv")
add_dependencies(Renderer VulkanShaders)
target_link_libraries(Renderer PRIVATE MyEngine)
target_compile_options(Renderer PRIVATE /W3 /utf-8 /MP)
set_target_properties(Renderer PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin" VS_DEBUGGER_WORKING_DIRECTORY "$<TARGET_FILE_DIR:Renderer>")
add_custom_command(TARGET Renderer POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/Assets" "$<TARGET_FILE_DIR:Renderer>/Assets")
add_custom_command(TARGET Renderer POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory "${VULKAN_SHADER_OUTPUT}" "$<TARGET_FILE_DIR:Renderer>/Assets/Shaders/Vulkan")
set_property(DIRECTORY PROPERTY VS_STARTUP_PROJECT Renderer)
file(GLOB EDITOR_SOURCES CONFIGURE_DEPENDS Editor/*.cpp)
add_executable(Editor ${EDITOR_SOURCES})
target_link_libraries(Editor PRIVATE MyEngine comctl32 comdlg32)
target_compile_options(Editor PRIVATE /W3 /utf-8 /MP)
target_link_options(Editor PRIVATE /MANIFEST:EMBED "/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/Editor/Editor.manifest")
set_target_properties(Editor PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin" VS_DEBUGGER_WORKING_DIRECTORY "$<TARGET_FILE_DIR:Editor>")
add_dependencies(Editor Renderer)
install(TARGETS Editor RUNTIME DESTINATION .)
install(TARGETS Renderer RUNTIME DESTINATION .)
install(DIRECTORY Assets DESTINATION .)
install(DIRECTORY "${VULKAN_SHADER_OUTPUT}/" DESTINATION Assets/Shaders/Vulkan)
install(FILES ThirdParty/LICENSES.txt ThirdParty/packages.json DESTINATION Licenses)
install(FILES ThirdParty/Vulkan/packages.json DESTINATION Licenses/Vulkan)
install(FILES ThirdParty/Vulkan/LICENSE-Apache-2.0.txt ThirdParty/Vulkan/LICENSE-Headers.md ThirdParty/Vulkan/LICENSE-MIT.txt DESTINATION Licenses/Vulkan)
install(FILES ThirdParty/Vulkan/Tools/LICENSE-LLVM.txt ThirdParty/Vulkan/Tools/LICENSE-MIT.txt ThirdParty/Vulkan/Tools/LICENSE-MS.txt DESTINATION Licenses/Vulkan/DXC)
install(FILES README.md DESTINATION .)
install(FILES cmake/Run-OpenGL.cmd cmake/Run-DX11.cmd DESTINATION .)
install(FILES Run-Sponza.cmd Run-Sibenik.cmd DESTINATION .)
install(FILES cmake/Run-Editor.cmd cmake/Run-DX12.cmd cmake/Run-Vulkan.cmd DESTINATION .)
install(FILES cmake/Run-Highlands.cmd DESTINATION .)
